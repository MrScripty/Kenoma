def edgeMatrix (v : Fin 4 → Fin 3 → ℝ) : Matrix (Fin 3) (Fin 3) ℝ :=
  fun i j => v j.succ i - v 0 i

theorem edge_translation_invariant (v : Fin 4 → Fin 3 → ℝ) (t : Fin 3 → ℝ) :
    edgeMatrix (fun j i => v j i + t i) = edgeMatrix v := by
  apply Matrix.ext
  intro i j
  simp [edgeMatrix]

theorem determinant_composition (A B : Matrix (Fin 3) (Fin 3) ℝ) :
    Matrix.det (A * B) = Matrix.det A * Matrix.det B := by
  exact Matrix.det_mul A B

def axialMatrix (lambda b : ℝ) : Matrix (Fin 3) (Fin 3) ℝ :=
  Matrix.diagonal ![lambda, b, b]

theorem axial_determinant (lambda b : ℝ) :
    Matrix.det (axialMatrix lambda b) = lambda * b * b := by
  simp [axialMatrix, Matrix.det_diagonal, Fin.prod_univ_succ, mul_assoc]

theorem isochoric_sqrt_construction (lambda : ℝ) (hlambda : 0 < lambda) :
    0 < 1 / Real.sqrt lambda ∧
    Matrix.det (axialMatrix lambda (1 / Real.sqrt lambda)) = 1 := by
  constructor
  · exact one_div_pos.mpr (Real.sqrt_pos.mpr hlambda)
  · rw [axial_determinant]
    calc
      lambda * (1 / Real.sqrt lambda) * (1 / Real.sqrt lambda) =
          lambda * (Real.sqrt lambda * Real.sqrt lambda)⁻¹ := by
        simp only [one_div, mul_assoc, mul_inv]
      _ = 1 := by
        rw [Real.mul_self_sqrt hlambda.le]
        exact mul_inv_cancel₀ (ne_of_gt hlambda)

/- Exact identities for Lab 5's declared split energy, per reference volume. -/
def shapeDensity (mu J I1 : ℝ) : ℝ :=
  mu / 2 * (J ^ (-2 / 3 : ℝ) * I1 - 3)

def volumeDensity (bulk J : ℝ) : ℝ := bulk / 2 * (J - 1) ^ 2

theorem dilation_shape_zero (mu s : ℝ) (hs : 0 < s) :
    shapeDensity mu (s ^ (3 : ℕ)) (3 * s ^ (2 : ℕ)) = 0 := by
  have hpow : (s ^ (3 : ℕ)) ^ (-2 / 3 : ℝ) = (s ^ (2 : ℕ))⁻¹ := by
    calc
      (s ^ (3 : ℕ)) ^ (-2 / 3 : ℝ) = s ^ (-2 : ℝ) := by
        rw [← Real.rpow_natCast_mul hs.le 3]
        congr 1
        norm_num
      _ = (s ^ (2 : ℕ))⁻¹ := by
        rw [show (-2 : ℝ) = -(2 : ℝ) by norm_num, Real.rpow_neg hs.le]
  unfold shapeDensity
  rw [hpow]
  have hn : s ^ (2 : ℕ) ≠ 0 := pow_ne_zero _ (ne_of_gt hs)
  have hcancel : (s ^ (2 : ℕ))⁻¹ * (3 * s ^ (2 : ℕ)) = 3 := by
    rw [mul_comm 3, ← mul_assoc, inv_mul_cancel₀ hn, one_mul]
  rw [hcancel]
  simp

def shearMatrix (gamma : ℝ) : Matrix (Fin 3) (Fin 3) ℝ :=
  !![1, gamma, 0; 0, 1, 0; 0, 0, 1]

def squaredInvariant (A : Matrix (Fin 3) (Fin 3) ℝ) : ℝ :=
  ∑ i, ∑ j, (A i j) ^ 2

theorem simple_shear_split (mu bulk gamma : ℝ) :
    Matrix.det (shearMatrix gamma) = 1 ∧
    shapeDensity mu (Matrix.det (shearMatrix gamma))
      (squaredInvariant (shearMatrix gamma)) = mu / 2 * gamma ^ 2 ∧
    volumeDensity bulk (Matrix.det (shearMatrix gamma)) = 0 := by
  have hdet : Matrix.det (shearMatrix gamma) = 1 := by
    simp [shearMatrix, Matrix.det_fin_three]
  have hI : squaredInvariant (shearMatrix gamma) = 3 + gamma ^ 2 := by
    simp [squaredInvariant, shearMatrix, Fin.sum_univ_succ]
    ring
  rw [hdet, hI]
  simp [shapeDensity, volumeDensity]

/- For F=diag(t,h,t), these are the actual split Cauchy components. -/
def lateralCauchy (mu bulk t h : ℝ) : ℝ :=
  mu / 3 * (t ^ (2 : ℕ) * h) ^ (-5 / 3 : ℝ) * (t ^ 2 - h ^ 2) +
    bulk * (t ^ 2 * h - 1)

def axialCauchy (mu bulk t h : ℝ) : ℝ :=
  2 * mu / 3 * (t ^ (2 : ℕ) * h) ^ (-5 / 3 : ℝ) * (h ^ 2 - t ^ 2) +
    bulk * (t ^ 2 * h - 1)

theorem mean_pressure_split (mu bulk t h : ℝ) :
    -(2 * lateralCauchy mu bulk t h + axialCauchy mu bulk t h) / 3 =
      bulk * (1 - t ^ 2 * h) := by
  unfold lateralCauchy axialCauchy
  ring

theorem free_side_plate_pressure (mu bulk t h : ℝ)
    (hfree : lateralCauchy mu bulk t h = 0) :
    -axialCauchy mu bulk t h = 3 * bulk * (1 - t ^ 2 * h) := by
  have hm := mean_pressure_split mu bulk t h
  rw [hfree] at hm
  linarith

end
